HashMap<Integer, Integer> freq = new HashMap<>();
for (int i = 0; i < A.length; i++) {
if (freq.containsKey(A[i]))
freq.put(A[i], freq.get(A[i]) + 1);
else
freq.put(A[i], 1);
}
int count = 0;
for (int key : freq.keySet()) {
if (freq.get(key) == 1) count++;
}
return count;
